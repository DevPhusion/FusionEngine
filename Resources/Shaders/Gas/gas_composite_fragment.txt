#version 330 core
in vec2 TexCoord;
out vec4 FragColor;

uniform sampler2D densityTex;
uniform vec4 coreColor;
uniform vec4 edgeColor;
uniform float threshold;
uniform float edgeSoft;
uniform vec2 texelSize;
uniform float time;
uniform float noiseScale;
uniform float noiseStrength;
uniform float riseSpeed;

float hash(vec2 p) {
    p = fract(p * vec2(123.34, 456.21));
    p += dot(p, p + 45.32);
    return fract(p.x * p.y);
}

float noise(vec2 p) {
    vec2 i = floor(p);
    vec2 f = fract(p);
    float a = hash(i);
    float b = hash(i + vec2(1.0, 0.0));
    float c = hash(i + vec2(0.0, 1.0));
    float d = hash(i + vec2(1.0, 1.0));
    vec2 u = f * f * (3.0 - 2.0 * f);
    return mix(a, b, u.x) + (c - a) * u.y * (1.0 - u.x) + (d - b) * u.x * u.y;
}

float fbm(vec2 p) {
    float value = 0.0;
    float amp = 0.5;
    for (int i = 0; i < 4; i++) {
        value += amp * noise(p);
        p *= 2.0;
        amp *= 0.5;
    }
    return value;
}

void main() {
    float d = texture(densityTex, TexCoord).r;
    if (d <= 0.001) discard;

    vec2 drift = vec2(0.0, time * riseSpeed);
    float n = fbm(TexCoord * noiseScale + drift);
    n += 0.5 * fbm(TexCoord * noiseScale * 2.3 - drift * 1.7);
    n /= 1.5;

    float carved = d * mix(1.0 - noiseStrength, 1.0 + noiseStrength * 0.3, n);

    float alpha = smoothstep(0.0, threshold * 1.4, carved);
    alpha *= smoothstep(0.0, edgeSoft * 3.0, d);
    if (alpha <= 0.001) discard;

    float densityLerp = clamp(carved / max(threshold * 1.6, 0.001), 0.0, 1.0);
    vec3 col = mix(edgeColor.rgb, coreColor.rgb, densityLerp);
    col *= 0.85 + 0.3 * n;

    float a = mix(edgeColor.a, coreColor.a, densityLerp) * alpha;
    FragColor = vec4(col, a);
}

#version 330 core
layout(location = 0) in vec3 aPos;
layout(location = 1) in vec2 aUV;
layout(location = 2) in vec4 aVertColor;

uniform mat4 projection;
uniform mat4 view;

out vec2 UV;
out vec4 VertColor;

void main() {
    UV = aUV;
    VertColor = aVertColor;
    gl_Position = projection * view * vec4(aPos, 1.0);
}
